dir: /pm-cli

-----create market
cargo run -- create-market \
    --market-id 1 \
    --metadata "Will Bitcoin reach $100k by end of 2024?" \
    --end-time $(($(date +%s) + 86400))

-----list markets
cargo run -- list-markets

-----resolve market
cargo run -- resolve-market --market-id 1 --outcome yes
